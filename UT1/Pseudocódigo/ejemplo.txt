// Escribe un programa que pida una cantidad en euros y la convierta a dólares (1 € = 1.15 $).

INICIO
  DEFINE cantidad COMO FLOAT
  DEFINE resultado COMO FLOAT
  ESCRIBE "Introduzca la cantidad en euros"
  LEER cantidad
  resultado <- CANTIDAD * 1.15
  ESCRIBE "La cantidad introducida en dólares es ", resultado
FIN
