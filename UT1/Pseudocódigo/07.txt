//  Realiza un programa que pida el precio de un producto sin IVA y el porcentaje de IVA que se le debe aplicar, expresado en formato decimal (por ejemplo, 0.21 para un 21 %). Calcula y muestra:
//  - El importe correspondiente al IVA.
//  - El precio final del producto con IVA incluido.

INICIO
  DEFINE precioBase, iva, importeIva, precioConIva COMO FLOAT
  ESCRIBIR "Introduce el precio sin IVA"
  LEER precioBase
  ESCRIBIR "Introduce el porcentaje de IVA en decimal"
  LEER iva
  importeIva <- precioBase * iva
  precioConIva <- precioBase + importeIva
  ESCRIBIR "El importe de IVA es ", importeIva
  ESCRIBIR "El precio con IVA incluido es ", precioConIva
FIN
